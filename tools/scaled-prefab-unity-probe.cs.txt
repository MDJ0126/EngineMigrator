using System;using System.IO;using System.Linq;using System.Collections.Generic;using System.Reflection;using UnityEngine;using UnityEditor;using UnityEditor.SceneManagement;using UnityEngine.SceneManagement;
public class EMScaledPrefabObserver:MonoBehaviour {
 public Component driver;public int fps;public float initial;private int last;private List<string> rows=new List<string>();private List<string> clocks=new List<string>();
 static string Read(object o,string n){object v=o.GetType().GetField(n).GetValue(o);return v is bool b?(b?"1":"0"):Convert.ToString(v,System.Globalization.CultureInfo.InvariantCulture);}
 void Awake(){Time.timeScale=initial;Time.captureDeltaTime=1f/fps;Time.fixedDeltaTime=.02f;}
 void LateUpdate(){try{int frame=int.Parse(Read(driver,"Frame"));if(frame<=last)return;last=frame;var row=new List<string>();foreach(var n in new[]{"Frame","Scale","Delta","Distinct","TemplateQuiet","ReturnedAwakes","ReturnedStarts","ReturnedSleepAwakes"})row.Add(Read(driver,n));foreach(var slot in new[]{"A","B","C","Copy","Sleep"}){var w=(Component)driver.GetType().GetField(slot).GetValue(driver);row.Add(w?"1":"0");if(w){var t=w.GetType();var peer=(Component)t.GetField("Peer").GetValue(w);var self=(GameObject)t.GetField("Self").GetValue(w);var target=(GameObject)t.GetField("Target").GetValue(w);bool refs=peer&&(Component)t.GetField("Peer").GetValue(peer)==w&&self==w.gameObject&&target==peer.gameObject;row.Add(w.gameObject.activeSelf?"1":"0");row.Add(w.gameObject.activeInHierarchy?"1":"0");row.Add(refs?"1":"0");}else{row.Add("-1");row.Add("-1");row.Add("-1");}foreach(var who in new[]{w,w?(Component)w.GetType().GetField("Peer").GetValue(w):null})foreach(var n in new[]{"Awakes","Enables","Starts","Updates","Fixeds","Disables","Id","AwakeScale","StartScale","Scale","Delta","Elapsed","RefsInAwake"})row.Add(who?Read(who,n):"-1");}rows.Add("["+string.Join(",",row)+"]");var rational=Time.timeAsRational;var capture=Time.captureDeltaTimeRational;clocks.Add("["+frame+","+rational.Count+","+rational.Ticks.Numerator+","+rational.Ticks.Denominator+","+capture.Count+","+Time.fixedTimeAsDouble.ToString("R",System.Globalization.CultureInfo.InvariantCulture)+"]");Time.captureDeltaTime=1f/fps;if(frame>=24){File.WriteAllText(Environment.GetEnvironmentVariable("EM_SCALED_PREFAB_OUTPUT")+".clock.json","["+string.Join(",",clocks)+"]");File.WriteAllText(Environment.GetEnvironmentVariable("EM_SCALED_PREFAB_OUTPUT"),"{\"status\":\"CAPTURED\",\"rows\":["+string.Join(",",rows)+"]}");Time.timeScale=1;EditorApplication.Exit(0);}}catch(Exception e){Debug.LogException(e);EditorApplication.Exit(1);}}
}
public static class EMScaledPrefabProbe {
 static BindingFlags flags=BindingFlags.Static|BindingFlags.NonPublic;
 static GameObject Prefab(string name,Type type,bool active){string folder="Assets/EngineMigratorScaledPrefabProbe/";var root=new GameObject(name);root.SetActive(active);var child=new GameObject("Prefab child");child.transform.SetParent(root.transform,false);child.transform.localPosition=new Vector3(1,2,3);var a=root.AddComponent(type);var b=child.AddComponent(type);foreach(var c in new[]{a,b}){type.GetField("Self").SetValue(c,c.gameObject);type.GetField("Target").SetValue(c,c==a?child:root);type.GetField("Peer").SetValue(c,c==a?b:a);type.GetField("Id").SetValue(c,c==a?7:9);}var asset=PrefabUtility.SaveAsPrefabAsset(root,folder+name+".prefab");UnityEngine.Object.DestroyImmediate(root);if(!asset)throw new Exception("Actual prefab asset save failed");return asset;}
 static void ExportUnit(string folder,string kind,string path,string name,Scene scene){var type=typeof(EngineMigratorBridge);var values=new Dictionary<string,object>{{"PackageRoot",folder},{"ProjectExportKind",kind},{"ProjectExportPath",path},{"ProjectExportName",name},{"ProjectExportScene",scene}};var old=values.Keys.ToDictionary(k=>k,k=>type.GetField(k,flags).GetValue(null));try{foreach(var e in values)type.GetField(e.Key,flags).SetValue(null,e.Value);Directory.CreateDirectory(folder);type.GetMethod("Export",flags).Invoke(null,null);}finally{foreach(var e in old)type.GetField(e.Key,flags).SetValue(null,e.Value);}}
 static void RunImported(Type driverType,int fps,float initial){
 string source=Path.GetFullPath(Path.Combine(Environment.GetEnvironmentVariable("EM_SCALED_PREFAB_EXPORT_ROOT"),"../Reverse"));string run=Guid.NewGuid().ToString("N").Substring(0,8);string destination=Path.GetFullPath(Path.Combine(Environment.GetEnvironmentVariable("EM_SCALED_PREFAB_EXPORT_ROOT"),"../RestoredInput_"+run));Directory.CreateDirectory(destination);var manifest=JsonUtility.FromJson<EMProjectPackage>(File.ReadAllText(Path.Combine(source,"project-package.json")));var ids=manifest.units.ToDictionary(u=>u.id,u=>u.id+"_"+run);
 foreach(var file in Directory.GetFiles(source,"*.json",SearchOption.AllDirectories)){string text=File.ReadAllText(file);foreach(var entry in ids)text=text.Replace(entry.Key,entry.Value);foreach(var name in new[]{"ScaledPrefabDriver","ScaledPrefabWorker"})text=System.Text.RegularExpressions.Regex.Replace(text,"\\b"+name+"\\b","Restored"+name);string target=Path.Combine(destination,file.Substring(source.Length).TrimStart(Path.DirectorySeparatorChar,Path.AltDirectorySeparatorChar));Directory.CreateDirectory(Path.GetDirectoryName(target));File.WriteAllText(target,text);}
 Debug.Log("SCALED_PREFAB_IMPORT_ROOT="+destination);foreach(var unit in JsonUtility.FromJson<EMProjectPackage>(File.ReadAllText(Path.Combine(destination,"project-package.json"))).units)Debug.Log("SCALED_PREFAB_IMPORT_PATH="+Path.GetFullPath(Path.Combine(destination,unit.package,"package.json"))+";SEP="+Path.DirectorySeparatorChar);var method=typeof(EngineMigratorBridge).GetMethod("ImportProject",flags);var packageRoot=typeof(EngineMigratorBridge).GetField("PackageRoot",flags);var prior=packageRoot.GetValue(null);string marker=Path.GetFullPath(".EngineMigratorProject.json");byte[] before=File.Exists(marker)?File.ReadAllBytes(marker):null;try{packageRoot.SetValue(null,destination);method.Invoke(null,null);}finally{packageRoot.SetValue(null,prior);if(before==null){if(File.Exists(marker))File.Delete(marker);}else File.WriteAllBytes(marker,before);}
 var imported=JsonUtility.FromJson<EMProjectImportResult>(File.ReadAllText(Path.Combine(destination,"unity-import-result.json")));if(imported.behaviourInstancesVerified!=5)throw new Exception("Actual prefab/project import component verification incomplete");var driver=UnityEngine.Object.FindObjectsByType(driverType,FindObjectsInactive.Include,FindObjectsSortMode.None).Cast<Component>().Single();MonoImporter.SetExecutionOrder(MonoScript.FromMonoBehaviour((MonoBehaviour)driver),-100);File.WriteAllText(Environment.GetEnvironmentVariable("EM_SCALED_PREFAB_OUTPUT")+".import.json",File.ReadAllText(Path.Combine(destination,"unity-import-result.json")));var observer=new GameObject("Imported prefab observer").AddComponent<EMScaledPrefabObserver>();observer.driver=driver;observer.fps=fps;observer.initial=initial;Time.timeScale=initial;Time.fixedDeltaTime=.02f;Time.captureDeltaTime=1f/fps;EditorSettings.enterPlayModeOptionsEnabled=true;EditorSettings.enterPlayModeOptions=EnterPlayModeOptions.DisableDomainReload|EnterPlayModeOptions.DisableSceneReload;var deadline=EditorApplication.timeSinceStartup+120;EditorApplication.update+=()=>{if(EditorApplication.timeSinceStartup>deadline)EditorApplication.Exit(2);};EditorApplication.EnterPlaymode();
 }
 public static void Run(){try{bool restored=Environment.GetEnvironmentVariable("EM_SCALED_PREFAB_RESTORED")=="1";float initial=float.Parse(Environment.GetEnvironmentVariable("EM_SCALED_PREFAB_INITIAL")??"0.5",System.Globalization.CultureInfo.InvariantCulture);int fps=int.Parse(Environment.GetEnvironmentVariable("EM_SCALED_PREFAB_FPS"));Time.timeScale=initial;Time.fixedDeltaTime=.02f;Time.captureDeltaTime=1f/fps;string prefix=restored?"Restored":"";var asm=typeof(ScaledPrefabDriver).Assembly;var workerType=asm.GetType(prefix+"ScaledPrefabWorker");var driverType=asm.GetType(prefix+"ScaledPrefabDriver");if(restored&&Directory.Exists(Path.Combine(Environment.GetEnvironmentVariable("EM_SCALED_PREFAB_EXPORT_ROOT"),"../Reverse"))){RunImported(driverType,fps,initial);return;}EditorSceneManager.NewScene(NewSceneSetup.EmptyScene,NewSceneMode.Single);var active=Prefab(prefix+"ScaledActive",workerType,true);var dormant=Prefab(prefix+"ScaledDormant",workerType,false);var scene=EditorSceneManager.NewScene(NewSceneSetup.EmptyScene,NewSceneMode.Single);var driver=new GameObject("Scaled prefab driver").AddComponent(driverType);driverType.GetField("ActivePrefab").SetValue(driver,active);driverType.GetField("DormantPrefab").SetValue(driver,dormant);MonoImporter.SetExecutionOrder(MonoScript.FromMonoBehaviour((MonoBehaviour)driver),-100);string scenePath="Assets/EngineMigratorScaledPrefabProbe/"+prefix+"Start.unity";if(!EditorSceneManager.SaveScene(scene,scenePath))throw new Exception("Prefab fixture scene save failed");
 if(!restored){string exportRoot=Environment.GetEnvironmentVariable("EM_SCALED_PREFAB_EXPORT_ROOT");var manifest=new EMProjectPackage();foreach(var pair in new[]{new{name="active",asset=active},new{name="dormant",asset=dormant}}){string path=AssetDatabase.GetAssetPath(pair.asset);string id=(string)typeof(EngineMigratorBridge).GetMethod("ProjectAssetId",flags).Invoke(null,new object[]{path,"p_"});var contents=PrefabUtility.LoadPrefabContents(path);try{ExportUnit(Path.Combine(exportRoot,pair.name),"prefab",path,pair.name,contents.scene);}finally{PrefabUtility.UnloadPrefabContents(contents);}manifest.units.Add(new EMProjectUnit{id=id,kind="prefab",package=pair.name,path=path,name=pair.name});}EditorSceneManager.OpenScene(scenePath);ExportUnit(Path.Combine(exportRoot,"scene"),"scene",scenePath,"ScaledPrefabStart",SceneManager.GetActiveScene());manifest.units.Add(new EMProjectUnit{id="scene_scaled_prefab",kind="scene",package="scene",path=scenePath,name="ScaledPrefabStart",enabled=true});File.WriteAllText(Path.Combine(exportRoot,"project-package.json"),JsonUtility.ToJson(manifest,true));driver=UnityEngine.Object.FindObjectsByType(driverType,FindObjectsInactive.Include,FindObjectsSortMode.None).Cast<Component>().Single();}
 var observer=new GameObject("Prefab observer").AddComponent<EMScaledPrefabObserver>();observer.driver=driver;observer.fps=fps;observer.initial=initial;Time.timeScale=initial;Time.fixedDeltaTime=.02f;Time.captureDeltaTime=1f/fps;EditorSettings.enterPlayModeOptionsEnabled=true;EditorSettings.enterPlayModeOptions=EnterPlayModeOptions.DisableDomainReload|EnterPlayModeOptions.DisableSceneReload;var deadline=EditorApplication.timeSinceStartup+120;EditorApplication.update+=()=>{if(EditorApplication.timeSinceStartup>deadline)EditorApplication.Exit(2);};EditorApplication.EnterPlaymode();}catch(Exception e){Debug.LogException(e);EditorApplication.Exit(1);}}
}
